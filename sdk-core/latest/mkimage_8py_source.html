12def boot_sector():
13 bs = bytearray(SECTOR)
14 bs[0:3] = b"\xEB\x3C\x90"
15 bs[3:11] = b"RAMDISK "
16 struct.pack_into("<HBHBHHBHHHII", bs, 11,
17 SECTOR, # bytes per sector
18 1, # sectors per cluster
19 1, # reserved sectors (the boot sector)
20 2, # FAT copies
21 16, # root directory entries (one sector)
22 TOTAL, # total sectors
23 0xF8, # media descriptor
24 1, # sectors per FAT
25 32, # sectors per track
26 2, # heads
27 0, # hidden sectors
28 0) # total sectors (32-bit field, unused)
29 struct.pack_into("<BBBI", bs, 36, 0x80, 0, 0x29, 0x52414D44)
30 bs[43:54] = b"RAMDISK "
31 bs[54:62] = b"FAT12 "
32 bs[510:512] = b"\x55\xAA"
33 return bytes(bs)

36def fat():
37 # Entries 0 and 1 are reserved; entry 2 (HELLO.TXT's only cluster) is end of chain.
38 f = bytearray(SECTOR)
39 f[0:5] = b"\xF8\xFF\xFF\xFF\x0F"
40 return bytes(f)

43def root_dir():
44 d = bytearray(SECTOR)
45 d[0:11] = b"RAMDISK "
46 d[11] = 0x08
47 entry = bytearray(32)
48 entry[0:11] = b"HELLO TXT"
49 entry[11] = 0x20
50 struct.pack_into("<HHHI", entry, 22, 0, 0x5B50, 2, len(HELLO))
51 struct.pack_into("<H", entry, 16, 0x5B50)
52 struct.pack_into("<H", entry, 18, 0x5B50)
53 d[32:64] = entry
54 return bytes(d)
