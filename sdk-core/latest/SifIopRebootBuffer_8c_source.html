44{
45 if (imgdrv_offset_ioprpimg == 0 || imgdrv_offset_ioprpsiz == 0) {
46 int i;
47
48 for (i = 0; (unsigned int)i < size__imgdrv_irx; i += 4) {
49 if (*(u32 *)((&((unsigned char *)_imgdrv_irx)[i])) == 0xDEC1DEC1) {
50 imgdrv_offset_ioprpimg = i;

52 if (*(u32 *)((&((unsigned char *)_imgdrv_irx)[i])) == 0xDEC2DEC2) {

244/* Generate an IOPRP image that contains IOPBTCONF, if the original contains IOPBTCONF.
245 This is required because UDNL will only seach succeeding IOPRP images for modules specified within IOPBTCONF. */
246static int generateIOPBTCONF_img(void *output, const void *ioprp)
247{
248 int size, offset, fsize_rounded;
249 romdir_t *romdir;
250 const u8 *ptr_in;
251 u8 *ptr_out;
252
253 romdir = (romdir_t *)ioprp;
254 if (strcmp(romdir[0].name, "RESET") || strcmp(romdir[1].name, "ROMDIR"))
255 return -1;
256
257 // Now locate IOPBTCONF
258 size = 0;
259 offset = 0;
260 for (; romdir->name[0] != '\0'; romdir++, offset += fsize_rounded) {
261 fsize_rounded = (romdir->size + 15) & ~15;
262 if (strcmp(romdir->name, "IOPBTCONF") == 0) {
263 romdir->name[0] = 'X'; // Change 'IOPBTCONF' to 'XOPBTCONF', so that UDNL will not find the original.
264 // Copy IOPBTCONF into the new image.
265 size = romdir->size + sizeof(iopbtconf_img_base);
266 ptr_out = (u8 *)output;
267 ptr_in = (const u8 *)ioprp;
268 memcpy(ptr_out, &iopbtconf_img_base, sizeof(iopbtconf_img_base));
269 memcpy(ptr_out + sizeof(iopbtconf_img_base), &ptr_in[offset], romdir->size);
270 ((romdir_t *)ptr_out)[3].size = romdir->size; // Update the size of IOPBTCONF within the generated IOPRP image.
271 break;
272 }
273 }
274
275 return size;
276}
