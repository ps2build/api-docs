66/* The name field is 8 bytes: longer names are cut, shorter ones padded with zeros. */
67#define IRX_ASM_NAME(modname) \
68 ".set\t.Lirx_name_len, 0\n\t" \
69 ".irpc\tc, " #modname "\n\t" \
70 ".if .Lirx_name_len < 8\n\t" \
71 ".ascii\t\"\\c\"\n\t" \
72 ".endif\n\t" \
73 ".set\t.Lirx_name_len, .Lirx_name_len + 1\n\t" \
74 ".endr\n\t" \
75 ".if .Lirx_name_len < 8\n\t" \
76 ".fill\t8 - .Lirx_name_len, 1, 0\n\t" \
77 ".endif\n"

79#ifdef __clang__
80/* The GNU assembler reads the section name below as ".text" plus a comment. Clang takes it literally and
81 puts the table header in a section of its own, away from the stubs that follow it, so write it as assembly. */
82#define DECLARE_IMPORT_TABLE(modname, major, minor) \
83 __asm__ (".section\t.text\n\t" \
84 ".balign\t4\n\t" \
85 ".word\t" XSTR(IMPORT_MAGIC) ", 0\n\t" \
86 ".short\t" XSTR(IRX_VER(major, minor)) ", 0\n\t" \
87 IRX_ASM_NAME(modname));
88#else
89#define DECLARE_IMPORT_TABLE(modname, major, minor) \
90static struct irx_import_table _imp_##modname \
91 __attribute__((section(".text\n\t#"), unused))= { \
92 magic: IMPORT_MAGIC, version: IRX_VER(major, minor), \
93 name: #modname, };
94#endif
95
96// .word 0x03e00008 == jr $ra (return immediately), this value will be patched later
97#define DECLARE_IMPORT(ord, name) \
98 __asm__ (".section\t.text\n\t" \
99 ".globl\t"#name"\n\t"#name":\n\t" \
100 ".word 0x03e00008\n\t" \
101 ".word " STR(0x24000000|ord));
102
103#define END_IMPORT_TABLE \
104 __asm__ (".section\t.text\n\t.word\t0, 0");
