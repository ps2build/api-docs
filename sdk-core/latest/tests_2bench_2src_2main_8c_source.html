89{

91
92 *p++ = 0x1000000000008001ull | ((u64)vertices << 0);
93 *p++ = 0x5551ull;
94 for (u32 i = 0; i < vertices; i++) {
95 u32 x = (i * 37) & 0x1ff, y = (i * 91) & 0x1ff, z = i * 3;

107__attribute__((noinline)) static u32 bench_decompress(u8 *out, const u8 *in, u32 in_len)
108{
109 u8 *o = out;
110 const u8 *end = in + in_len;
111
112 while (in < end) {
113 u32 tag = *in++;
114
115 if (tag < 0x80) {
116 u32 n = tag + 1;
117
118 while (n--)
119 *o++ = *in++;
120 } else {
121 u32 len = (tag & 0x7f) + 3;
122 u32 dist = ((u32)in[0] << 8 | in[1]) + 1;
123 const u8 *m = o - dist;
124
125 in += 2;
126 while (len--)
127 *o++ = *m++;
128 }
129 }
130 return (u32)(o - out);
131}

206__attribute__((noinline)) static void bench_inflate_table(struct inf_huff *h, const u8 *lens, u32 n)
207{
208 u16 offs[16], next[16];
209 u32 code = 0;
210
211 memset(h->count, 0, sizeof(h->count));
212 memset(h->fast, 0, sizeof(h->fast));
213 for (u32 i = 0; i < n; i++)
214 h->count[lens[i]]++;
215 h->count[0] = 0;
216 offs[1] = 0;
217 for (u32 len = 1; len < 15; len++)
218 offs[len + 1] = offs[len] + h->count[len];
219 for (u32 len = 1; len < 16; len++) {
220 code = (code + h->count[len - 1]) << 1;
221 next[len] = (u16)code;
222 }
223 for (u32 i = 0; i < n; i++) {
224 u32 len = lens[i];
225
226 if (len == 0)
227 continue;
228 h->symbol[offs[len]++] = (u16)i;
229 if (len <= INF_FAST_BITS) {
230 u32 c = next[len], rev = 0;
231
232 for (u32 b = 0; b < len; b++)
233 rev |= ((c >> b) & 1) << (len - 1 - b);
234 for (u32 k = rev; k < (1u << INF_FAST_BITS); k += 1u << len)
235 h->fast[k] = (u16)((len << 9) | i);
236 }
237 next[len]++;
238 }

241static inline __attribute__((always_inline)) u32 inf_decode(struct inf_state *s, const struct inf_huff *h)
242{
243 u32 e, code = 0, first = 0, index = 0;
244
245 inf_refill(s);
246 e = h->fast[s->bitbuf & ((1u << INF_FAST_BITS) - 1)];
247 if (e) {
248 s->bitbuf >>= e >> 9;
249 s->bitcnt -= e >> 9;
250 return e & 0x1ff;
251 }
252 for (u32 len = 1; len < 16; len++) {
253 u32 count = h->count[len];
254
255 code |= s->bitbuf & 1;
256 s->bitbuf >>= 1;
257 s->bitcnt--;
258 if (code - first < count)
259 return h->symbol[index + code - first];
260 index += count;
261 first = (first + count) << 1;
262 code <<= 1;
263 }
264 return 0xffff;
265}

379__attribute__((noinline)) static float bench_physics(u32 frames)
380{
381 float energy = 0.0f;
382 const float dt = 1.0f / 60.0f;
383
384 for (u32 f = 0; f < frames; f++) {
385 for (u32 i = 0; i < PARTICLES; i++) {
386 vy[i] -= 9.8f * dt;
387 px[i] += vx[i] * dt;
388 py[i] += vy[i] * dt;
389 pz[i] += vz[i] * dt;
390 if (py[i] < 0.0f) {
391 py[i] = -py[i];
392 vy[i] = -vy[i] * 0.8f;
393 }
394 float d2 = px[i] * px[i] + pz[i] * pz[i];
395 if (d2 > 100.0f) {
396 float inv = 1.0f / sqrtf(d2);
397 vx[i] = -px[i] * inv;
398 vz[i] = -pz[i] * inv;
399 }
400 }
401 }
402 for (u32 i = 0; i < PARTICLES; i++)
403 energy += vx[i] * vx[i] + vy[i] * vy[i] + vz[i] * vz[i];
404 return energy;
405}
